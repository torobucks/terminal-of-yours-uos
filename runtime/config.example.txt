# Terminal-of-Yours 配置示例
# 复制为 runtime/config.txt 生效（不复制则用默认值，本文件仅作参考）
#
# shell: 终端 shell。
#   auto（默认）——
#     Windows：优先 pwsh（PowerShell 7），找不到回退 powershell.exe（PowerShell 5.1）
#     Linux/macOS（UOS 等）：优先用户默认登录 shell（$SHELL），其次 bash → zsh → sh，
#       即原生 Linux 终端；装了 pwsh 也可用 shell=pwsh 显式切换
#   显式指定（不探测）：pwsh / powershell.exe / /bin/bash / /usr/bin/zsh 等
shell=auto

# port: 监听端口（默认 8787，被占自动 +1，实际端口写入 runtime/toy.port）
port=8787

# buffer: 断线重连回放缓冲上限（字节，默认 524288 = 512KB）
buffer=524288

# runTimeout: toy.sh run 默认超时（秒，默认 120）
runTimeout=120

# noProfile: 1 时不加载 shell profile（PowerShell profile / .bashrc / .zshrc；默认 0 = 加载）
noProfile=0
